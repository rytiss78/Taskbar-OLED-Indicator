cmake_minimum_required(VERSION 3.24)
project(TaskbarOLEDIndicator LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Build with MSVC on Windows x64.")
endif()
add_library(AcRes SHARED src/oled_shim.cpp src/exports.def)
target_compile_features(AcRes PRIVATE cxx_std_17)
target_compile_definitions(AcRes PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(AcRes PRIVATE dbghelp user32 advapi32 windowsapp)
set_target_properties(AcRes PROPERTIES PREFIX "" OUTPUT_NAME "AcRes")
